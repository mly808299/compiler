// --- SECTION 1: Variables & Math Operations ---
var a int = 10;
var b int = 20;
var result int = 0;

ADD result a b;      // result = 10 + 20 = 30
PLE result 5;        // result += 5 -> 35
MIE result 3;        // result -= 3 -> 32
INC result;          // result++ -> 33

print(result);       // Expected Output: 33


// --- SECTION 2: Type Casting (to_int, to_float) ---
var pi float = 3.99;
var pi_int int = to_int(pi); // Should be 3
print(pi_int);

var flag bool = true;
var flag_val int = to_int(flag); // Should be 1
print(flag_val);


// --- SECTION 3: Arrays & Loops ---
array my_list = [1, 2, 3, 4];
var sum_loop int = 0;

foreach (item in my_list) {
    PLE sum_loop item;  // sum = 1+2+3+4 = 10
}
print(sum_loop);


// --- SECTION 4: Pattern Matching (Match) ---
var status int = 2;

match status {
    1 -> print("Status is One"),
    2 -> print("Status is Two"),   // This should execute
    _ -> print("Status is Default")
}
